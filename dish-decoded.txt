<template>
  <div class="app-container">
    <!-- 锟斤拷锟斤拷 -->
    <el-form :model="queryParams" ref="queryRef" :inline="true" v-show="showSearch" label-width="90px">
      <el-form-item label="锟斤拷品锟斤拷锟斤拷" prop="dishName">
        <el-input v-model="queryParams.dishName" placeholder="锟斤拷锟斤拷锟斤拷锟狡凤拷锟斤拷锟? clearable style="width: 200px" @keyup.enter="handleQuery" />
      </el-form-item>
      <el-form-item label="锟斤拷锟斤拷" prop="categoryId">
        <el-select v-model="queryParams.categoryId" placeholder="锟斤拷选锟斤拷锟斤拷锟? clearable style="width: 200px">
          <el-option v-for="c in categoryOptions" :key="c.categoryId" :label="c.categoryName" :value="c.categoryId" />
        </el-select>
      </el-form-item>
      <el-form-item label="状态" prop="status">
        <el-select v-model="queryParams.status" placeholder="状态" clearable style="width: 160px">
          <el-option label="锟较硷拷" value="1" />
          <el-option label="锟铰硷拷" value="0" />
        </el-select>
      </el-form-item>
      <el-form-item>
        <el-button type="primary" icon="Search" @click="handleQuery">锟斤拷锟斤拷</el-button>
        <el-button icon="Refresh" @click="resetQuery">锟斤拷锟斤拷</el-button>
      </el-form-item>
    </el-form>

    <!-- 锟斤拷锟竭帮拷钮 -->
    <el-row :gutter="10" class="mb8">
      <el-col :span="1.5">
        <el-button type="primary" plain icon="Plus" v-hasPermi="['takeout:dish:add']" @click="handleAdd">锟斤拷锟斤拷</el-button>
      </el-col>
      <el-col :span="1.5">
        <el-button type="danger" plain icon="Delete" :disabled="multiple" v-hasPermi="['takeout:dish:remove']" @click="handleDelete">删锟斤拷</el-button>
      </el-col>
      <right-toolbar v-model:showSearch="showSearch" @queryTable="getList" />
    </el-row>

    <!-- 锟斤拷锟斤拷 -->
    <el-table v-loading="loading" :data="dataList" @selection-change="handleSelectionChange">
      <el-table-column type="selection" width="55" align="center" />
      <el-table-column label="锟斤拷锟? prop="dishId" width="80" />
      <el-table-column label="锟斤拷品锟斤拷锟斤拷" prop="dishName" />
      <el-table-column label="锟斤拷锟斤拷" prop="categoryName" width="120" />
      <el-table-column label="图片" width="80" align="center">
        <template #default="scope">
          <el-image v-if="scope.row.image" :src="scope.row.image" fit="cover" style="width:50px;height:50px;border-radius:4px" />
          <span v-else>-</span>
        </template>
      </el-table-column>
      <el-table-column label="锟桔革拷" prop="price" width="80" align="right">
        <template #default="scope">? {{ scope.row.price }}</template>
      </el-table-column>
      <el-table-column label="锟斤拷锟? prop="stock" width="80" align="center" />
      <el-table-column label="锟斤拷锟斤拷" prop="sales" width="80" align="center" />
      <el-table-column label="状态" width="80" align="center">
        <template #default="scope">
          <el-switch v-model="scope.row.status" active-value="1" inactive-value="0" @change="(val) => handleStatusChange(scope.row, val)" />
        </template>
      </el-table-column>
      <el-table-column label="锟斤拷锟斤拷时锟斤拷" prop="createTime" width="170" />
      <el-table-column label="锟斤拷锟斤拷" width="180" align="center">
        <template #default="scope">
          <el-button link type="primary" v-hasPermi="['takeout:dish:edit']" @click="handleUpdate(scope.row)">锟睫革拷</el-button>
          <el-button link type="primary" v-hasPermi="['takeout:dish:remove']" @click="handleDelete(scope.row)">删锟斤拷</el-button>
        </template>
      </el-table-column>
    </el-table>

    <pagination v-show="total" :total="total" v-model:page="queryParams.pageNum" v-model:limit="queryParams.pageSize" @pagination="getList" />

    <!-- 锟斤拷锟斤拷/锟睫革拷 锟斤拷锟斤拷 -->
    <el-dialog :title="title" v-model="open" width="560px" append-to-body>
      <el-form ref="formRef" :model="form" :rules="rules" label-width="100px">
        <el-form-item label="锟斤拷品锟斤拷锟斤拷" prop="dishName">
          <el-input v-model="form.dishName" placeholder="锟斤拷锟斤拷锟斤拷锟狡凤拷锟斤拷锟? />
        </el-form-item>
        <el-form-item label="锟斤拷锟斤拷锟斤拷锟斤拷" prop="categoryId">
          <el-select v-model="form.categoryId" placeholder="锟斤拷选锟斤拷锟斤拷锟? style="width: 100%">
            <el-option v-for="c in categoryOptions" :key="c.categoryId" :label="c.categoryName" :value="c.categoryId" />
          </el-select>
        </el-form-item>
        <el-form-item label="锟斤拷品图片" prop="image">
          <image-upload v-model="form.image" :fileType="['png','jpg','jpeg']" :fileSize="20" />
        </el-form-item>
        <el-form-item label="锟桔革拷" prop="price">
          <el-input-number v-model="form.price" :min="0" :precision="2" :step="1" controls-position="right" style="width: 160px" />
          <span style="margin-left:8px;color:#909399">元</span>
        </el-form-item>
        <el-form-item label="锟斤拷锟? prop="stock">
          <el-input-number v-model="form.stock" :min="0" controls-position="right" style="width: 160px" />
        </el-form-item>
        <el-form-item label="状态">
          <el-radio-group v-model="form.status">
            <el-radio value="1">锟较硷拷</el-radio>
            <el-radio value="0">锟铰硷拷</el-radio>
          </el-radio-group>
        </el-form-item>
        <el-form-item label="锟斤拷锟斤拷" prop="description">
          <el-input v-model="form.description" type="textarea" :rows="2" placeholder="锟斤拷锟斤拷锟斤拷锟斤拷锟斤拷" />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button type="primary" @click="submitForm">确 锟斤拷</el-button>
        <el-button @click="cancel">取 锟斤拷</el-button>
      </template>
    </el-dialog>
  </div>
</template>

<script setup name="Dish">
import { ref, reactive, getCurrentInstance, onMounted } from 'vue'
import { listDish, addDish, updateDish, delDish, changeDishStatus } from '@/api/takeout/dish'
import { optionselect as listCategoryOptions } from '@/api/takeout/dishCategory'

const { proxy } = getCurrentInstance()

const dataList = ref([])
const categoryOptions = ref([])
const open = ref(false)
const loading = ref(true)
const showSearch = ref(true)
const total = ref(0)
const multiple = ref(true)
const title = ref('')
const ids = ref([])
const single = ref(false)

const data = reactive({
  form: {},
  queryParams: { pageNum: 1, pageSize: 10, dishName: undefined, categoryId: undefined, status: undefined },
  rules: {
    dishName: [{ required: true, message: '锟斤拷品锟斤拷锟狡诧拷锟斤拷为锟斤拷', trigger: 'blur' }],
    categoryId: [{ required: true, message: '锟斤拷选锟斤拷锟斤拷锟?, trigger: 'change' }],
    price: [{ required: true, message: '锟斤拷锟斤拷锟斤拷鄹锟?, trigger: 'blur' }],
    stock: [{ required: true, message: '锟斤拷锟斤拷锟斤拷锟斤拷', trigger: 'blur' }]
  }
})
const { queryParams, form, rules } = data

function getList() {
  loading.value = true
  listDish(queryParams).then(res => { dataList.value = res.rows; total.value = res.total; loading.value = false })
}
function getCategoryOptions() {
  listCategoryOptions().then(res => { categoryOptions.value = res.data || [] })
}

function handleQuery() { queryParams.pageNum = 1; getList() }
function resetQuery() { proxy.resetForm('queryRef'); handleQuery() }

function handleSelectionChange(sel) {
  ids.value = sel.map(i => i.dishId)
  multiple.value = sel.length === 0
  single.value = sel.length !== 1
}

function handleAdd() { reset(); open.value = true; title.value = '锟斤拷锟斤拷锟斤拷品' }
function handleUpdate(row) { reset(); form.value = { ...row }; open.value = true; title.value = '锟睫改诧拷品' }
function submitForm() {
  proxy.$refs['formRef'].validate(valid => {
    if (!valid) return
    if (form.value.dishId != null) {
      updateDish(form.value).then(() => { proxy.$modal.msgSuccess('锟睫改成癸拷'); open.value = false; getList() })
    } else {
      addDish(form.value).then(() => { proxy.$modal.msgSuccess('锟斤拷锟斤拷锟缴癸拷'); open.value = false; getList() })
    }
  })
}
function cancel() { open.value = false; reset() }
function reset() {
  form.value = { dishId: null, dishName: '', categoryId: undefined, image: '', price: 0, stock: 0, status: '1', description: '' }
  proxy.resetForm('formRef')
}

function handleStatusChange(row, val) {
  const text = val === '1' ? '锟较硷拷' : '锟铰硷拷'
  proxy.$modal.confirm(`确锟较斤拷锟斤拷品锟斤拷${row.dishName}锟斤拷${text}锟斤拷`).then(() => changeDishStatus(row.dishId, val))
    .then(() => { proxy.$modal.msgSuccess(`${text}锟缴癸拷`); getList() })
    .catch(() => { row.status = val === '1' ? '0' : '1' })
}

function handleDelete(row) {
  const toIds = row?.dishId ? [row.dishId] : ids.value
  proxy.$modal.confirm(`确锟斤拷删锟斤拷锟斤拷锟轿拷锟?{toIds.join(',')}锟斤拷锟侥诧拷品锟斤拷`).then(() => delDish(toIds))
    .then(() => { getList(); proxy.$modal.msgSuccess('删锟斤拷锟缴癸拷') }).catch(() => {})
}

onMounted(() => { getList(); getCategoryOptions() })
</script>
