<template>
  <div class="app-container">
    <!-- 查询区 -->
    <el-form :model="queryParams" ref="queryRef" :inline="true" v-show="showSearch" label-width="80px">
      <el-form-item label="商家名称" prop="merchantName">
        <el-input
          v-model="queryParams.merchantName"
          placeholder="请输入商家名称"
          clearable
          style="width: 240px"
          @keyup.enter="handleQuery"
        />
      </el-form-item>
      <el-form-item label="联系电话" prop="contactPhone">
        <el-input
          v-model="queryParams.contactPhone"
          placeholder="请输入联系电话"
          clearable
          style="width: 240px"
          @keyup.enter="handleQuery"
        />
      </el-form-item>
      <el-form-item label="营业状态" prop="status">
        <el-select v-model="queryParams.status" placeholder="商家状态" clearable style="width: 240px">
          <el-option
            v-for="dict in takeout_business_status"
            :key="dict.value"
            :label="dict.label"
            :value="dict.value"
          />
        </el-select>
      </el-form-item>
      <el-form-item label="审核状态" prop="auditStatus">
        <el-select v-model="queryParams.auditStatus" placeholder="审核状态" clearable style="width: 240px">
          <el-option
            v-for="dict in takeout_audit_status"
            :key="dict.value"
            :label="dict.label"
            :value="dict.value"
          />
        </el-select>
      </el-form-item>
      <el-form-item>
        <el-button type="primary" icon="Search" @click="handleQuery">搜索</el-button>
        <el-button icon="Refresh" @click="resetQuery">重置</el-button>
      </el-form-item>
    </el-form>

    <!-- 工具栏 -->
    <el-row :gutter="10" class="mb8">
      <el-col :span="1.5">
        <el-button
          type="primary"
          plain
          icon="Plus"
          @click="handleAdd"
          v-hasPermi="['takeout:merchant:add']"
        >新增</el-button>
      </el-col>
      <el-col :span="1.5">
        <el-button
          type="success"
          plain
          icon="Edit"
          :disabled="single"
          @click="handleUpdate"
          v-hasPermi="['takeout:merchant:edit']"
        >修改</el-button>
      </el-col>
      <el-col :span="1.5">
        <el-button
          type="danger"
          plain
          icon="Delete"
          :disabled="multiple"
          @click="handleDelete"
          v-hasPermi="['takeout:merchant:remove']"
        >删除</el-button>
      </el-col>
      <el-col :span="1.5">
        <el-button
          type="warning"
          plain
          icon="Download"
          @click="handleExport"
          v-hasPermi="['takeout:merchant:export']"
        >导出</el-button>
      </el-col>
      <right-toolbar v-model:showSearch="showSearch" @queryTable="getList" :columns="columns" storageKey="takeout_merchant" />
    </el-row>

    <!-- 表格 -->
    <el-table v-loading="loading" :data="merchantList" @selection-change="handleSelectionChange">
      <el-table-column type="selection" width="50" align="center" />
      <el-table-column label="商家编号" align="center" key="merchantId" prop="merchantId" v-if="columns.merchantId.visible" />
      <el-table-column label="商家名称" align="center" key="merchantName" prop="merchantName" v-if="columns.merchantName.visible" :show-overflow-tooltip="true" />
      <el-table-column label="联系人" align="center" key="contactName" prop="contactName" v-if="columns.contactName.visible" />
      <el-table-column label="联系电话" align="center" key="contactPhone" prop="contactPhone" v-if="columns.contactPhone.visible" width="120" />
      <el-table-column label="商家地址" align="center" key="address" prop="address" v-if="columns.address.visible" :show-overflow-tooltip="true" />
      <el-table-column label="营业状态" align="center" key="status" v-if="columns.status.visible">
        <template #default="scope">
          <el-switch
            v-model="scope.row.status"
            active-value="0"
            inactive-value="1"
            @change="handleStatusChange(scope.row)"
            v-hasPermi="['takeout:merchant:status']"
          ></el-switch>
        </template>
      </el-table-column>
      <el-table-column label="审核状态" align="center" key="auditStatus" v-if="columns.auditStatus.visible">
        <template #default="scope">
          <dict-tag :options="takeout_audit_status" :value="scope.row.auditStatus" />
        </template>
      </el-table-column>
      <el-table-column label="创建时间" align="center" prop="createTime" v-if="columns.createTime.visible" width="160">
        <template #default="scope">
          <span>{{ parseTime(scope.row.createTime) }}</span>
        </template>
      </el-table-column>
      <el-table-column label="操作" align="center" width="220" class-name="small-padding fixed-width">
        <template #default="scope">
          <el-button link type="primary" icon="Edit" @click="handleUpdate(scope.row)" v-hasPermi="['takeout:merchant:edit']">修改</el-button>
          <el-button link type="primary" icon="Delete" @click="handleDelete(scope.row)" v-hasPermi="['takeout:merchant:remove']">删除</el-button>
          <el-button link type="primary" icon="CircleCheck" @click="openAuditDialog(scope.row)" v-hasPermi="['takeout:merchant:audit']" v-if="scope.row.auditStatus === '0'">审核</el-button>
        </template>
      </el-table-column>
    </el-table>

    <pagination
      v-show="total > 0"
      :total="total"
      v-model:page="queryParams.pageNum"
      v-model:limit="queryParams.pageSize"
      @pagination="getList"
    />

    <!-- 新增 / 修改 对话框 -->
    <el-dialog :title="title" v-model="open" width="600px" append-to-body>
      <el-form :model="form" :rules="rules" ref="merchantRef" label-width="100px">
        <el-row>
          <el-col :span="12">
            <el-form-item label="商家名称" prop="merchantName">
              <el-input v-model="form.merchantName" placeholder="请输入商家名称" maxlength="100" />
            </el-form-item>
          </el-col>
          <el-col :span="12">
            <el-form-item label="联系人" prop="contactName">
              <el-input v-model="form.contactName" placeholder="请输入联系人" maxlength="50" />
            </el-form-item>
          </el-col>
        </el-row>
        <el-row>
          <el-col :span="12">
            <el-form-item label="联系电话" prop="contactPhone">
              <el-input v-model="form.contactPhone" placeholder="请输入联系电话" maxlength="20" />
            </el-form-item>
          </el-col>
          <el-col :span="12">
            <el-form-item label="绑定用户" prop="userId">
              <el-select
                v-model="form.userId"
                placeholder="不绑定则留空"
                clearable
                filterable
                style="width: 100%"
                @clear="form.userId = null"
              >
                <el-option
                  v-for="u in userOptions"
                  :key="u.userId"
                  :label="u.nickName + '（' + u.userName + '）'"
                  :value="u.userId"
                />
              </el-select>
            </el-form-item>
          </el-col>
        </el-row>
        <el-row>
          <el-col :span="24">
            <el-form-item label="商家地址" prop="address">
              <el-input v-model="form.address" placeholder="请输入商家地址" maxlength="255" />
            </el-form-item>
          </el-col>
        </el-row>
        <el-row>
          <el-col :span="12">
            <el-form-item label="营业状态">
              <el-radio-group v-model="form.status">
                <el-radio
                  v-for="dict in takeout_business_status"
                  :key="dict.value"
                  :value="dict.value"
                >{{ dict.label }}</el-radio>
              </el-radio-group>
            </el-form-item>
          </el-col>
        </el-row>
        <el-row>
          <el-col :span="24">
            <el-form-item label="备注" prop="remark">
              <el-input v-model="form.remark" type="textarea" placeholder="请输入内容" maxlength="500" />
            </el-form-item>
          </el-col>
        </el-row>
      </el-form>
      <template #footer>
        <div class="dialog-footer">
          <el-button type="primary" @click="submitForm">确 定</el-button>
          <el-button @click="cancel">取 消</el-button>
        </div>
      </template>
    </el-dialog>

    <!-- 审核 对话框 -->
    <el-dialog title="商家审核" v-model="auditOpen" width="500px" append-to-body>
      <el-form :model="auditForm" :rules="auditRules" ref="auditRef" label-width="100px">
        <el-form-item label="审核结果" prop="auditStatus">
          <el-radio-group v-model="auditForm.auditStatus">
            <el-radio
              v-for="dict in takeout_audit_status.filter(d => d.value !== '0')"
              :key="dict.value"
              :value="dict.value"
            >{{ dict.label }}</el-radio>
          </el-radio-group>
        </el-form-item>
        <el-form-item label="审核备注" prop="auditRemark">
          <el-input v-model="auditForm.auditRemark" type="textarea" placeholder="请输入审核备注（驳回时必填）" maxlength="255" />
        </el-form-item>
      </el-form>
      <template #footer>
        <div class="dialog-footer">
          <el-button type="primary" @click="submitAudit">确 定</el-button>
          <el-button @click="auditOpen = false">取 消</el-button>
        </div>
      </template>
    </el-dialog>
  </div>
</template>

<script setup name="TakeoutMerchant">
import { listMerchant, getMerchant, addMerchant, updateMerchant, delMerchant, changeMerchantStatus, auditMerchant, exportMerchant } from "@/api/takeout/merchant"
import { optionselectUser } from "@/api/system/user"

const { proxy } = getCurrentInstance()
const { takeout_business_status, takeout_audit_status } = useDict("takeout_business_status", "takeout_audit_status")

// 列表数据
const merchantList = ref([])
const open = ref(false)
const loading = ref(true)
const showSearch = ref(true)
const ids = ref([])
const single = ref(true)
const multiple = ref(true)
const total = ref(0)
const title = ref("")

// 审核弹窗
const auditOpen = ref(false)
const auditForm = ref({ merchantId: undefined, auditStatus: '1', auditRemark: '' })

// 绑定用户下拉选项
const userOptions = ref([])

// 列显隐
const columns = ref({
  merchantId:   { label: '商家编号', visible: true },
  merchantName: { label: '商家名称', visible: true },
  contactName:  { label: '联系人',   visible: true },
  contactPhone: { label: '联系电话', visible: true },
  address:      { label: '商家地址', visible: true },
  status:       { label: '营业状态', visible: true },
  auditStatus:  { label: '审核状态', visible: true },
  createTime:   { label: '创建时间', visible: true }
})

const data = reactive({
  // 查询条件
  queryParams: {
    pageNum: 1,
    pageSize: 10,
    merchantName: undefined,
    contactPhone: undefined,
    status: undefined,
    auditStatus: undefined
  },
  // 表单
  form: {},
  // 校验规则
  rules: {
    merchantName: [
      { required: true, message: "商家名称不能为空", trigger: "blur" },
      { min: 2, max: 100, message: "商家名称长度必须介于 2 和 100 之间", trigger: "blur" }
    ],
    contactPhone: [
      { pattern: /^1[3|4|5|6|7|8|9][0-9]\d{8}$/, message: "请输入正确的手机号码", trigger: "blur" }
    ]
  },
  // 审核校验
  auditRules: {
    auditStatus: [{ required: true, message: "请选择审核结果", trigger: "change" }],
    auditRemark: [
      {
        validator: (rule, value, callback) => {
          if (auditForm.value.auditStatus === '2' && !value) {
            callback(new Error("驳回时必须填写审核备注"))
          } else {
            callback()
          }
        },
        trigger: "blur"
      }
    ]
  }
})

const { queryParams, form, rules, auditRules } = toRefs(data)

/** 加载用户下拉选项 */
function loadUserOptions() {
  optionselectUser().then(res => {
    userOptions.value = res.data || []
  })
}

/** 查询商家列表 */
function getList() {
  loading.value = true
  listMerchant(queryParams.value).then(res => {
    loading.value = false
    merchantList.value = res.rows
    total.value = res.total
  })
}

/** 搜索按钮 */
function handleQuery() {
  queryParams.value.pageNum = 1
  getList()
}

/** 重置搜索 */
function resetQuery() {
  proxy.resetForm("queryRef")
  handleQuery()
}

/** 多选框选中 */
function handleSelectionChange(selection) {
  ids.value = selection.map(item => item.merchantId)
  single.value = selection.length != 1
  multiple.value = !selection.length
}

/** 重置表单 */
function reset() {
  form.value = {
    merchantId: undefined,
    merchantName: undefined,
    contactName: undefined,
    contactPhone: undefined,
    address: undefined,
    userId: undefined,
    status: '0',
    remark: undefined
  }
  proxy.resetForm("merchantRef")
}

/** 新增按钮 */
function handleAdd() {
  reset()
  open.value = true
  title.value = "添加商家"
  loadUserOptions()
}

/** 修改按钮 */
function handleUpdate(row) {
  reset()
  const merchantId = row.merchantId || ids.value
  getMerchant(merchantId).then(res => {
    form.value = res.data
    open.value = true
    title.value = "修改商家"
    loadUserOptions()
  })
}

/** 提交表单 */
function submitForm() {
  proxy.$refs["merchantRef"].validate(valid => {
    if (valid) {
      // 未选择绑定用户时显式置为null，确保后端能清空 user_id
      if (form.value.userId === undefined || form.value.userId === '') {
        form.value.userId = null
      }
      if (form.value.merchantId != undefined) {
        updateMerchant(form.value).then(() => {
          proxy.$modal.msgSuccess("修改成功")
          open.value = false
          getList()
        })
      } else {
        addMerchant(form.value).then(() => {
          proxy.$modal.msgSuccess("新增成功")
          open.value = false
          getList()
        })
      }
    }
  })
}

/** 删除按钮 */
function handleDelete(row) {
  const merchantIds = row.merchantId || ids.value
  proxy.$modal.confirm('是否确认删除商家编号为 "' + merchantIds + '"的数据项？').then(() => {
    return delMerchant(merchantIds)
  }).then(() => {
    getList()
    proxy.$modal.msgSuccess("删除成功")
  }).catch(() => {})
}

/** 营业状态切换 */
function handleStatusChange(row) {
  const text = row.status === "0" ? "营业中" : "已打烊"
  proxy.$modal.confirm('确认要切换商家 "' + row.merchantName + '" 为 "' + text + '" 吗？').then(() => {
    return changeMerchantStatus(row.merchantId, row.status)
  }).then(() => {
    proxy.$modal.msgSuccess("状态切换成功")
  }).catch(() => {
    row.status = row.status === "0" ? "1" : "0"
  })
}

/** 打开审核弹窗 */
function openAuditDialog(row) {
  auditForm.value = {
    merchantId: row.merchantId,
    auditStatus: '1',
    auditRemark: ''
  }
  auditOpen.value = true
}

/** 提交审核 */
function submitAudit() {
  proxy.$refs["auditRef"].validate(valid => {
    if (valid) {
      auditMerchant(auditForm.value).then(() => {
        proxy.$modal.msgSuccess("审核成功")
        auditOpen.value = false
        getList()
      })
    }
  })
}

/** 导出按钮 */
function handleExport() {
  proxy.download("takeout/merchant/export", { ...queryParams.value }, `merchant_${new Date().getTime()}.xlsx`)
}

/** 取消按钮 */
function cancel() {
  open.value = false
  reset()
}

onMounted(() => {
  getList()
})
</script>
